cmake_minimum_required(VERSION 3.20)
project(SwarmMod LANGUAGES C CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "SwarmMod requires MSVC Win32 (-A Win32)")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SOKULIB_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/deps/SokuLib" CACHE PATH "SokuLib source checkout")
if(NOT EXISTS "${SOKULIB_ROOT}/CMakeLists.txt")
  message(FATAL_ERROR "Set SOKULIB_ROOT to a SokuDev/SokuLib checkout")
endif()
add_subdirectory("${SOKULIB_ROOT}" sokulib EXCLUDE_FROM_ALL)
include(cmake/Dependencies.cmake)
add_library(swarmmod MODULE src/main.cpp src/State.hpp src/Publisher.hpp)
target_link_libraries(swarmmod PRIVATE SokuLib CURL::libcurl winhttp crypt32)
target_compile_definitions(swarmmod PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(swarmmod PRIVATE /utf-8 /W4)
set_target_properties(swarmmod PROPERTIES PREFIX "" OUTPUT_NAME "SwarmMod")
enable_testing()
add_executable(swarmmod_state_tests tests/state_tests.cpp)
target_compile_options(swarmmod_state_tests PRIVATE /utf-8 /W4)
add_test(NAME swarmmod_state COMMAND swarmmod_state_tests)
add_executable(swarmmod_http_tests EXCLUDE_FROM_ALL tests/http_tests.cpp)
target_link_libraries(swarmmod_http_tests PRIVATE SokuLib CURL::libcurl winhttp crypt32)
target_compile_definitions(swarmmod_http_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(swarmmod_http_tests PRIVATE /utf-8)
add_executable(swarmmod_tls_probe EXCLUDE_FROM_ALL tests/tls_probe.cpp)
target_link_libraries(swarmmod_tls_probe PRIVATE SokuLib CURL::libcurl winhttp crypt32)
target_compile_definitions(swarmmod_tls_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(swarmmod_tls_probe PRIVATE /utf-8)
